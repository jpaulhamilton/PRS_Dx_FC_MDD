############################################################
# Split-half exploratory behavioral correlation analysis
#
# MDD-main-effect connectivity clusters
#
# Discovery:
#   Spearman correlation, two-tailed, uncorrected p < .05
#
# Replication:
#   Only correlations significant in discovery are tested
#
# PRS analysis:
#   HighPRS + LowPRS collapsed within each MDD group
#
# Six collapsed groups x two sexes:
#   females_CTL
#   females_cMDD
#   females_rMDD
#   males_CTL
#   males_cMDD
#   males_rMDD
#
# Two connectivity definitions: big and small
############################################################


############################
# 1. DIRECTORIES
############################

GROUP_DIR <- file.path(BASE_DIR, "groups")
CON_DIR   <- file.path(BASE_DIR, "DMN_LN_CON")
BEH_DIR   <- file.path(BASE_DIR, "BEHAVE_EXTENDED")

RESULT_DIR <- file.path(BASE_DIR, "RESULTS_EXTENDED")

if (!dir.exists(RESULT_DIR)) {
    dir.create(RESULT_DIR, recursive = TRUE)
}


############################
# 2. SETTINGS
############################

# Connectivity definitions to analyse
CON_TYPES <- c("big", "small")

# Discovery threshold
DISCOVERY_P <- 0.05

# Replication threshold
REPLICATION_P <- 0.05


############################
# 3. BEHAVIOR FILES
############################

behavior_files <- list.files(
    BEH_DIR,
    pattern = "\\.csv$",
    full.names = TRUE
)

cat("\nBehavior files found:\n")

for (f in behavior_files) {
    cat("  ", basename(f), "\n")
}

if (length(behavior_files) == 0) {
    stop("No behavioral CSV files found in BEHAVE directory.")
}


############################
# 4. READ BEHAVIOR DATA
############################

behavior_data <- list()

for (f in behavior_files) {

    beh_name <- tools::file_path_sans_ext(basename(f))

    dat <- read.csv(
        f,
        stringsAsFactors = FALSE
    )

    # Make sure ID is character
    if (!"eid" %in% names(dat)) {
        stop(
            "Behavior file does not contain 'eid': ",
            f
        )
    }

    dat$eid <- as.character(dat$eid)


    ############################
    # SPECIAL CASE: NEUROTICISM
    ############################

    if (beh_name == "neuroticism") {

        # Count the number of "1" responses across
        # behave1 ... behave12 for each subject.
        #
        # -1 and -3 are therefore automatically excluded
        # because only values equal to 1 are counted.

        behave_columns <- grep(
            "^behave[0-9]+$",
            names(dat),
            value = TRUE
        )

        if (length(behave_columns) == 0) {
            stop(
                "Neuroticism file does not contain behave1 ... behave12: ",
                f
            )
        }

        dat$neuroticism <- apply(
            dat[, behave_columns, drop = FALSE],
            1,
            function(x) {
                sum(x == 1, na.rm = TRUE)
            }
        )

        behavior_data[[beh_name]] <- dat[, c("eid", beh_name)]

        cat(
            "Loaded behavior:",
            beh_name,
            "(counted number of 1s)",
            "N =",
            nrow(dat),
            "\n"
        )


    ############################
    # ALL OTHER BEHAVIORS
    ############################

    } else {

        ####################################################
        # All recreated behavior files use:
        #
        #   eid,value
        ####################################################

        if (!"value" %in% names(dat)) {
            stop(
                "Behavior file does not contain 'value': ",
                f
            )
        }

        value_column <- "value"


        ############################
        # CONVERT TO NUMERIC
        ############################

        dat[[value_column]] <- as.numeric(
            dat[[value_column]]
        )


        ############################
        # REMOVE UKB MISSING CODES
        ############################

        # UK Biobank:
        # Negative values are treated as missing.
        #
        # IMPORTANT:
        # Zeroes are retained.

        dat[[value_column]][
            !is.na(dat[[value_column]]) &
            dat[[value_column]] < 0
        ] <- NA


        ############################
        # SPECIAL CASE: 20018
        ############################

        if (beh_name == "field_20018") {

            # Swap:
            #   1 -> 2
            #   2 -> 1
            #
            # Use indices so the replacement is simultaneous
            # and does not accidentally undo itself.

            idx1 <- !is.na(dat[[value_column]]) &
                    dat[[value_column]] == 1

            idx2 <- !is.na(dat[[value_column]]) &
                    dat[[value_column]] == 2

            dat[[value_column]][idx1] <- 2
            dat[[value_column]][idx2] <- 1

            cat(
                "Loaded behavior:",
                beh_name,
                "(removed negative values; swapped 1 <-> 2)",
                "N =",
                nrow(dat),
                "\n"
            )


        ############################
        # SPECIAL CASE: REACTION TIME
        ############################

        } else if (beh_name == "reacton_time") {

            # Remove reaction times <= 100 ms.
            # Keep only values > 100 ms.

            dat[[value_column]][
                !is.na(dat[[value_column]]) &
                dat[[value_column]] <= 100
            ] <- NA

            cat(
                "Loaded behavior:",
                beh_name,
                "(removed negative values and reaction times <= 100 ms)",
                "N =",
                nrow(dat),
                "\n"
            )


        ############################
        # ALL OTHER VARIABLES
        ############################

        } else {

            cat(
                "Loaded behavior:",
                beh_name,
                "(removed negative values; zeroes retained)",
                "N =",
                nrow(dat),
                "\n"
            )
        }


        ############################
        # RENAME BEHAVIOR VARIABLE
        ############################

        names(dat)[
            names(dat) == value_column
        ] <- beh_name

        behavior_data[[beh_name]] <-
            dat[, c("eid", beh_name)]
    }
}

get_connectivity_file <- function(sex, con_type) {

    if (sex == "female") {

        f <- file.path(
            CON_DIR,
            paste0(
                "clust_table_female_n5_2_clusternum_merge_",
                con_type,
                ".csv"
            )
        )

    } else {

        f <- file.path(
            CON_DIR,
            paste0(
                "clust_table_male_n5_2_clusternum_merge_",
                con_type,
                ".csv"
            )
        )
    }

    if (!file.exists(f)) {
        stop("Connectivity file not found: ", f)
    }

    dat <- read.csv(
        f,
        stringsAsFactors = FALSE
    )

    dat$SUBJ <- as.character(dat$SUBJ)

    dat
}


############################
# 7. FUNCTION TO READ
#    ONE GROUP
############################

get_group_data <- function(group_file,
                           connectivity,
                           behave_all) {

    groups <- read.csv(
        group_file,
        stringsAsFactors = FALSE
    )

    groups$SUBJ <- as.character(groups$SUBJ)

    # Match connectivity
    dat <- merge(
        groups,
        connectivity[, c("SUBJ", "CON")],
        by = "SUBJ",
        all.x = TRUE
    )

    # Match behaviors
    dat <- merge(
        dat,
        behave_all,
        by.x = "SUBJ",
        by.y = "eid",
        all.x = TRUE
    )

    dat
}


############################
# 8. SPEARMAN FUNCTION
############################

run_spearman <- function(dat, behavior) {

    x <- dat$CON
    y <- dat[[behavior]]

    # Keep complete pairs only
    keep <- complete.cases(x, y)

    x <- x[keep]
    y <- y[keep]

    n <- length(x)

    if (n < 4) {

        return(
            data.frame(
                N = n,
                rho = NA_real_,
                p = NA_real_
            )
        )
    }

    test <- suppressWarnings(
        cor.test(
            x,
            y,
            method = "spearman",
            exact = FALSE,
            alternative = "two.sided"
        )
    )

    data.frame(
        N = n,
        rho = unname(test$estimate),
        p = test$p.value
    )
}

            # DISCOVERY CORRELATIONS
            ############################

            discovery_group_results <- list()


            for (behavior in behavior_names) {

                result <- run_spearman(
                    discovery_data,
                    behavior
                )

                result$sex <- sex
                result$group <- group_name
                result$connectivity <- con_type
                result$behavior <- behavior
                result$analysis_type <- "PRS_collapsed"

                discovery_group_results[[behavior]] <- result


                cat(
                    "Discovery:",
                    behavior,
                    "rho =",
                    round(result$rho, 3),
                    "p =",
                    signif(result$p, 3),
                    "\n"
                )
            }


            discovery_group_results <- do.call(
                rbind,
                discovery_group_results
            )

            rownames(discovery_group_results) <- NULL


            ############################
            # COLUMN ORDER
            ############################

            discovery_group_results <- discovery_group_results[
                c(
                    "sex",
                    "analysis_type",
                    "group",
                    "connectivity",
                    "behavior",
                    "N",
                    "rho",
                    "p"
                )
            ]


            ############################
            # DISCOVERY SIGNIFICANT
            ############################

            discovery_sig <- discovery_group_results[
                !is.na(discovery_group_results$p) &
                discovery_group_results$p < DISCOVERY_P,
            ]


            cat(
                "\nDiscovery correlations p <",
                DISCOVERY_P,
                ":",
                nrow(discovery_sig),
                "\n"
            )


            ############################
            # REPLICATION
            #
            # Run ALL behaviors tested
            # in discovery.
            #
            # IMPORTANT:
            # replicated remains FALSE unless:
            #
            #   1. discovery p < .05
            #   2. replication p < .05
            #   3. same direction
            #
            # Thus this expands the replication
            # output without changing the definition
            # of a replicated finding.
            ############################

            replication_group_results <- list()


            for (behavior in behavior_names) {


                ############################
                # FIND DISCOVERY RESULT
                ############################

                discovery_row <- discovery_group_results[
                    discovery_group_results$behavior == behavior,
                ]


                ############################
                # DISCOVERY VALUES
                ############################

                discovery_N <- discovery_row$N

                discovery_rho <- discovery_row$rho

                discovery_p <- discovery_row$p


                ############################
                # REPLICATION
                ############################

                rep_result <- run_spearman(
                    replication_data,
                    behavior
                )


                replication_N <- rep_result$N

                replication_rho <- rep_result$rho

                replication_p <- rep_result$p


                ############################
                # SAME DIRECTION
                ############################

                same_direction <- NA


                if (!is.na(discovery_rho) &&
                    !is.na(replication_rho)) {

                    same_direction <-
                        sign(discovery_rho) ==
                        sign(replication_rho)
                }


                ############################
                # SIGNIFICANCE FLAGS
                ############################

                discovery_significant <- FALSE

                if (!is.na(discovery_p)) {

                    discovery_significant <-
                        discovery_p < DISCOVERY_P
                }


                replication_significant <- FALSE

                if (!is.na(replication_p)) {

                    replication_significant <-
                        replication_p < REPLICATION_P
                }


                ############################
                # REPLICATED
                #
                # SAME DEFINITION AS BEFORE
                ############################

                replicated <- FALSE


                if (!is.na(same_direction)) {

                    replicated <-
                        discovery_significant &&
                        replication_significant &&
                        same_direction
                }


                ############################
                # STORE RESULT
                ############################

                replication_group_results[[behavior]] <-
                    data.frame(

                        sex = sex,

                        analysis_type = "PRS_collapsed",

                        group = group_name,

                        connectivity = con_type,

                        behavior = behavior,

                        N_discovery =
                            discovery_N,

                        rho_discovery =
                            discovery_rho,

                        p_discovery =
                            discovery_p,

                        discovery_significant =
                            discovery_significant,

                        N_replication =
                            replication_N,

                        rho_replication =
                            replication_rho,

                        p_replication =
                            replication_p,

                        replication_significant =
                            replication_significant,

                        same_direction =
                            same_direction,

                        replicated =
                            replicated
                    )


                cat(
                    "Replication:",
                    behavior,
                    "rho =",
                    round(replication_rho, 3),
                    "p =",
                    signif(replication_p, 3),
                    "\n"
                )
            }


            ############################
            # COMBINE REPLICATION RESULTS
            ############################

            replication_group_results <- do.call(
                rbind,
                replication_group_results
            )

            rownames(replication_group_results) <- NULL


            ############################
            # COLUMN ORDER
            ############################

            replication_group_results <-
                replication_group_results[
                    c(
                        "sex",
                        "analysis_type",
                        "group",
                        "connectivity",
                        "behavior",
                        "N_discovery",
                        "rho_discovery",
                        "p_discovery",
                        "discovery_significant",
                        "N_replication",
                        "rho_replication",
                        "p_replication",
                        "replication_significant",
                        "same_direction",
                        "replicated"
                    )
                ]

            ############################
            # SAVE COLLAPSED RESULTS
            ############################

            discovery_filename <- paste0(
                group_name,
                "_",
                con_type,
                "_discovery.csv"
            )

            replication_filename <- paste0(
                group_name,
                "_",
                con_type,
                "_replication.csv"
            )

            write.csv(
                discovery_group_results,
                file.path(
                    RESULT_DIR,
                    discovery_filename
                ),
                row.names = FALSE
            )

            write.csv(
                replication_group_results,
                file.path(
                    RESULT_DIR,
                    replication_filename
                ),
                row.names = FALSE
            )


            ############################
            # ADD TO MASTER TABLES
            ############################

            all_discovery_results[[result_counter]] <-
                discovery_group_results

            all_replication_results[[result_counter]] <-
                replication_group_results

            result_counter <- result_counter + 1


            ############################
            # PRINT REPLICATION RESULTS
            ############################

            if (nrow(replication_group_results) > 0) {

                cat(
                    "\nREPLICATION RESULTS:\n"
                )

                print(
                    replication_group_results
                )

            } else {

                cat(
                    "\nNo discovery correlations passed p < ",
                    DISCOVERY_P,
                    "\n",
                    sep = ""
                )
            }

        }  # closes for (mdd_group in MDD_GROUPS)

    }  # closes for (sex in c("female", "male"))

}  # closes for (con_type in CON_TYPES)

# FUNCTION:
# READ ONE FULL PRS GROUP
# AND ADD BEHAVIOR DATA
############################

get_behavior_group_data <- function(
    group_file,
    behave_all
) {

    groups <- read.csv(
        group_file,
        stringsAsFactors = FALSE
    )

    groups$SUBJ <- as.character(
        groups$SUBJ
    )

    if (anyDuplicated(groups$SUBJ)) {

        stop(
            "Duplicate SUBJ values found in group file: ",
            group_file
        )
    }

    dat <- merge(
        groups[, "SUBJ", drop = FALSE],
        behave_all,
        by.x = "SUBJ",
        by.y = "eid",
        all.x = TRUE
    )

    dat
}

############################
# FUNCTION:
# MEDIAN + RAW MAD
############################

get_median_mad <- function(x) {

    x <- x[
        !is.na(x)
    ]

    if (length(x) == 0) {

        return(
            c(
                N = 0,
                median = NA_real_,
                MAD = NA_real_
            )
        )
    }

    med <- median(x)

    ############################
    # MEDIAN ABSOLUTE DEVIATION
    # AROUND THE MEDIAN
    ############################

    mad_raw <- median(
        abs(x - med)
    )

    c(
        N = length(x),
        median = med,
        MAD = mad_raw
    )
}

                    p_value = test$p.value,

                wilcox_counter + 1

                group_2,

get_full_connectivity_group <- function(
    sex,
    mdd_group,
    connectivity
) {

    sex_prefix <- ifelse(
        sex == "female",
        "females",
        "males"
    )


    ############################
    # FULL UNSPLIT GROUP FILES
    ############################

    file_high <- file.path(
        GROUP_DIR,
        paste0(
            sex_prefix,
            "_HighPRS_",
            mdd_group,
            ".csv"
        )
    )

    file_low <- file.path(
        GROUP_DIR,
        paste0(
            sex_prefix,
            "_LowPRS_",
            mdd_group,
            ".csv"
        )
    )


    ############################
    # CHECK FILES
    ############################

    if (!file.exists(file_high)) {
        stop(
            "Missing group file: ",
            file_high
        )
    }

    if (!file.exists(file_low)) {
        stop(
            "Missing group file: ",
            file_low
        )
    }


    ############################
    # READ GROUP MEMBERSHIP
    ############################

    high <- read.csv(
        file_high,
        stringsAsFactors = FALSE
    )

    low <- read.csv(
        file_low,
        stringsAsFactors = FALSE
    )

    high$SUBJ <- as.character(
        high$SUBJ
    )

    low$SUBJ <- as.character(
        low$SUBJ
    )


    ############################
    # COMBINE HIGH + LOW PRS
    ############################

    subjects <- rbind(
        high[, "SUBJ", drop = FALSE],
        low[, "SUBJ", drop = FALSE]
    )


    ############################
    # CHECK DUPLICATE SUBJECTS
    ############################

    if (anyDuplicated(subjects$SUBJ)) {

        stop(
            "Duplicate SUBJ values after combining HighPRS and LowPRS: ",
            sex,
            " ",
            mdd_group
        )
    }


    ############################
    # ADD CONNECTIVITY
    ############################

    dat <- merge(
        subjects,
        connectivity[, c("SUBJ", "CON")],
        by = "SUBJ",
        all.x = TRUE
    )

    dat
}
